# 数据库设计

> PostgreSQL + GORM，支持JSONB字段存储AC等结构化数据。

## 一、核心表结构

### 1.1 用户表 (users)

```sql
CREATE TABLE users (
    id BIGSERIAL PRIMARY KEY,
    username VARCHAR(100) UNIQUE NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    hashed_password VARCHAR(255) NOT NULL,
    role VARCHAR(20) NOT NULL CHECK (role IN ('product', 'developer', 'tester', 'admin')),
    avatar_url VARCHAR(500),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_users_role ON users(role);
```

**GORM模型**：
```go
type User struct {
    ID           uint      `gorm:"primaryKey" json:"id"`
    Username     string    `gorm:"size:100;uniqueIndex" json:"username"`
    Email        string    `gorm:"size:255;uniqueIndex" json:"email"`
    HashedPassword string  `gorm:"size:255;not null" json:"-"`
    Role         string    `gorm:"size:20;not null" json:"role"`
    AvatarURL    string    `gorm:"size:500" json:"avatar_url,omitempty"`
    CreatedAt    time.Time `json:"created_at"`
    UpdatedAt    time.Time `json:"updated_at"`
}
```

### 1.2 项目表 (projects)

```sql
CREATE TABLE projects (
    id BIGSERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    description TEXT,
    owner_id BIGINT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    agile_mode VARCHAR(20) NOT NULL DEFAULT 'kanban' CHECK (agile_mode IN ('scrum', 'kanban')),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_projects_owner ON projects(owner_id);
CREATE INDEX idx_projects_mode ON projects(agile_mode);
```

**GORM模型**：
```go
type Project struct {
    ID          uint      `gorm:"primaryKey" json:"id"`
    Name        string    `gorm:"size:255;not null" json:"name"`
    Description string    `gorm:"type:text" json:"description,omitempty"`
    OwnerID     uint      `gorm:"not null" json:"owner_id"`
    Owner       *User     `gorm:"foreignKey:OwnerID" json:"owner,omitempty"`
    AgileMode   string    `gorm:"size:20;not null;default:kanban" json:"agile_mode"`
    CreatedAt   time.Time `json:"created_at"`
    UpdatedAt   time.Time `json:"updated_at"`
}
```

### 1.3 项目成员表 (project_members)

```sql
CREATE TABLE project_members (
    id BIGSERIAL PRIMARY KEY,
    project_id BIGINT NOT NULL REFERENCES projects(id) ON DELETE CASCADE,
    user_id BIGINT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    role_in_project VARCHAR(20) NOT NULL CHECK (role_in_project IN ('product', 'developer', 'tester')),
    joined_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(project_id, user_id)
);

CREATE INDEX idx_project_members_project ON project_members(project_id);
CREATE INDEX idx_project_members_user ON project_members(user_id);
```

**GORM模型**：
```go
type ProjectMember struct {
    ID             uint      `gorm:"primaryKey" json:"id"`
    ProjectID      uint      `gorm:"not null" json:"project_id"`
    Project        *Project  `gorm:"foreignKey:ProjectID" json:"project,omitempty"`
    UserID         uint      `gorm:"not null" json:"user_id"`
    User           *User     `gorm:"foreignKey:UserID" json:"user,omitempty"`
    RoleInProject  string    `gorm:"size:20;not null" json:"role_in_project"`
    JoinedAt       time.Time `json:"joined_at"`
}
```

### 1.4 用户故事表 (user_stories)

**核心表，存储所有用户故事和验收标准**：

```sql
CREATE TABLE user_stories (
    id BIGSERIAL PRIMARY KEY,
    project_id BIGINT NOT NULL REFERENCES projects(id) ON DELETE CASCADE,
    title VARCHAR(500) NOT NULL,
    description TEXT,
    story_type VARCHAR(50) NOT NULL DEFAULT 'feature' CHECK (story_type IN ('feature', 'bug', 'chore')),
    status VARCHAR(50) NOT NULL DEFAULT 'backlog' CHECK (status IN ('backlog', 'ready', 'in_progress', 'test', 'done')),
    priority INT DEFAULT 0 CHECK (priority >= 0 AND priority <= 4),
    points INT CHECK (points IN (1, 2, 3, 5, 8, 13)),
    assigned_to BIGINT REFERENCES users(id) ON DELETE SET NULL,
    created_by BIGINT NOT NULL REFERENCES users(id),
    position FLOAT DEFAULT 0,
    acceptance_criteria JSONB DEFAULT '[]',
    tags JSONB DEFAULT '[]',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_stories_project ON user_stories(project_id);
CREATE INDEX idx_stories_status ON user_stories(status);
CREATE INDEX idx_stories_assigned ON user_stories(assigned_to);
CREATE INDEX idx_stories_priority ON user_stories(priority);
CREATE INDEX idx_stories_position ON user_stories(project_id, position);
```

**验收标准JSONB结构**：
```json
[
  {
    "id": "ac-1",
    "description": "用户可以输入邮箱和密码",
    "status": "passed",
    "order": 1
  },
  {
    "id": "ac-2",
    "description": "密码至少8位，包含字母和数字",
    "status": "pending",
    "order": 2
  }
]
```

**GORM模型**：
```go
type AcceptanceCriteria struct {
    ID          string `json:"id"`
    Description string `json:"description"`
    Status      string `json:"status"` // pending, passed, failed
    Order       int    `json:"order"`
}

type UserStory struct {
    ID                 uint                 `gorm:"primaryKey" json:"id"`
    ProjectID          uint                 `gorm:"not null" json:"project_id"`
    Project            *Project             `gorm:"foreignKey:ProjectID" json:"project,omitempty"`
    Title              string               `gorm:"size:500;not null" json:"title"`
    Description        string               `gorm:"type:text" json:"description,omitempty"`
    StoryType          string               `gorm:"size:50;not null;default:feature" json:"story_type"`
    Status             string               `gorm:"size:50;not null;default:backlog" json:"status"`
    Priority           int                  `gorm:"default:0" json:"priority"`
    Points             int                  `json:"points,omitempty"`
    AssignedTo         *uint                `json:"assigned_to,omitempty"`
    Assignee           *User                `gorm:"foreignKey:AssignedTo" json:"assignee,omitempty"`
    CreatedBy          uint                 `gorm:"not null" json:"created_by"`
    Creator            *User                `gorm:"foreignKey:CreatedBy" json:"creator,omitempty"`
    Position           float64              `gorm:"default:0" json:"position"`
    AcceptanceCriteria datatypes.JSON       `gorm:"type:jsonb;default:'[]'" json:"acceptance_criteria"`
    Tags               datatypes.JSON       `gorm:"type:jsonb;default:'[]'" json:"tags,omitempty"`
    CreatedAt          time.Time            `json:"created_at"`
    UpdatedAt          time.Time            `json:"updated_at"`
}
```

## 三、数据库迁移策略

使用GORM AutoMigrate或手动迁移文件：

> 当前仓库实际实现以 `internal/database/database.go` 中的 GORM `AutoMigrate` 为准；本章后续触发器、备份、监控内容属于设计建议，仓库内未提供对应 SQL/脚本。

```go
// migrations/migrate.go
func AutoMigrate(db *gorm.DB) error {
    return db.AutoMigrate(
        &User{},
        &Project{},
        &ProjectMember{},
        &UserStory{},
        &ActivityLog{},
        &Task{},
        &TestCase{},
        &BoardColumn{},
    )
}
```

## 五、数据完整性约束

### 5.1 状态流转触发器

```sql
CREATE OR REPLACE FUNCTION validate_status_transition()
RETURNS TRIGGER AS $$
BEGIN
    IF OLD.status IS DISTINCT FROM NEW.status THEN
        IF NOT EXISTS (
            SELECT 1 FROM unnest(ARRAY[
                ARRAY['backlog', 'ready'],
                ARRAY['backlog', 'in_progress'],
                ARRAY['ready', 'backlog'],
                ARRAY['ready', 'in_progress'],
                ARRAY['in_progress', 'ready'],
                ARRAY['in_progress', 'test'],
                ARRAY['in_progress', 'done'],
                ARRAY['test', 'in_progress'],
                ARRAY['test', 'done'],
                ARRAY['done', 'test']
            ]) AS allowed(allowed_from TEXT, allowed_to TEXT)
            WHERE allowed.allowed_from = OLD.status
              AND allowed.allowed_to = NEW.status
        ) THEN
            RAISE EXCEPTION 'Invalid status transition from % to %',
                OLD.status, NEW.status;
        END IF;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER validate_story_status_transition
BEFORE UPDATE OF status ON user_stories
FOR EACH ROW
EXECUTE FUNCTION validate_status_transition();
```

### 5.2 更新时间触发器

```sql
CREATE OR REPLACE FUNCTION update_updated_at()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- 为需要的表创建触发器
CREATE TRIGGER update_users_updated_at
BEFORE UPDATE ON users
FOR EACH ROW
EXECUTE FUNCTION update_updated_at();

CREATE TRIGGER update_projects_updated_at
BEFORE UPDATE ON projects
FOR EACH ROW
EXECUTE FUNCTION update_updated_at();

CREATE TRIGGER update_stories_updated_at
BEFORE UPDATE ON user_stories
FOR EACH ROW
EXECUTE FUNCTION update_updated_at();
```

```bash
# 每日全量备份
pg_dump -U postgres -d storybook > backup_$(date +%Y%m%d).sql

# 仅备份结构
pg_dump -U postgres -d storybook --schema-only > schema.sql
```

```bash

wal_level = replica
archive_mode = on
```

### 8.2 AC查询示例

```sql
-- 查找所有包含特定AC的故事
SELECT id, title
FROM user_stories
WHERE acceptance_criteria::jsonb @> '[{"description": "密码"}]';

-- 统计每个故事的AC完成情况
SELECT
    id,
    title,
    (acceptance_criteria->'metadata'->>'completion_percentage')::float AS ac_completion
FROM user_stories
WHERE acceptance_criteria::jsonb->'metadata' IS NOT NULL;

-- 更新特定AC状态
UPDATE user_stories
SET acceptance_criteria = jsonb_set(
    acceptance_criteria,
    '{acceptance_criteria,0,status}',
    '"passed"'
)
WHERE id = 123;
```

## 九、MCP数据库访问层

```go
// pkg/mcp/db_accessor.go
type MCPDatabaseAccessor struct {
    db *gorm.DB
}

// 获取用户故事及其AC
func (m *MCPDatabaseAccessor) GetUserStoryWithAC(storyID uint) (*UserStory, error) {
    var story UserStory
    err := m.db.Preload("Project").
        Preload("Assignee").
        Preload("Creator").
        First(&story, storyID).Error
    return &story, err
}

// 检查AC覆盖度
func (m *MCPDatabaseAccessor) CheckACCoverage(storyID uint) (*ACCoverage, error) {
    var story UserStory
    if err := m.db.First(&story, storyID).Error; err != nil {
        return nil, err
    }

    // 解析AC JSONB
    var criteria struct {
        AcceptanceCriteria []AcceptanceCriteria `json:"acceptance_criteria"`
        Metadata           struct {
            TotalAC              int     `json:"total_ac"`
            PassedAC             int     `json:"passed_ac"`
            PendingAC            int     `json:"pending_ac"`
            FailedAC             int     `json:"failed_ac"`
            CompletionPercentage float64 `json:"completion_percentage"`
        } `json:"metadata"`
    }

    if err := json.Unmarshal(story.AcceptanceCriteria, &criteria); err != nil {
        return nil, err
    }

    return &ACCoverage{
        StoryID:              storyID,
        TotalAC:              criteria.Metadata.TotalAC,
        PassedAC:             criteria.Metadata.PassedAC,
        PendingAC:            criteria.Metadata.PendingAC,
        FailedAC:             criteria.Metadata.FailedAC,
        CompletionPercentage: criteria.Metadata.CompletionPercentage,
    }, nil
}

// 更新AC状态
func (m *MCPDatabaseAccessor) UpdateACStatus(storyID uint, acID string, status string) error {
    var story UserStory
    if err := m.db.First(&story, storyID).Error; err != nil {
        return err
    }

    var criteria []AcceptanceCriteria
    if err := json.Unmarshal(story.AcceptanceCriteria, &criteria); err != nil {
        return err
    }

    for i, ac := range criteria {
        if ac.ID == acID {
            criteria[i].Status = status
            criteria[i].UpdatedAt = time.Now()
            break
        }
    }

    updated, err := json.Marshal(criteria)
    if err != nil {
        return err
    }

    return m.db.Model(&story).Update("acceptance_criteria", updated).Error
}
```
